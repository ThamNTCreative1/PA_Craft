    private void MovePlayer()
    {
        if(Input.GetKeyDown(KeyCode.Space))Jump();
        Vector2 input=Vector2.ClampMagnitude(joystick.Value+new Vector2(Input.GetAxisRaw("Horizontal"),Input.GetAxisRaw("Vertical")),1);
        if(input.sqrMagnitude>.01f && !sdkStarted){sdkStarted=true;Luna.Unity.LifeCycle.GameStarted();Track("first_move",1);hintHand.gameObject.SetActive(false);}
        Vector3 forward=gameCamera.transform.forward;forward.y=0;forward.Normalize();
        Vector3 right=gameCamera.transform.right;right.y=0;right.Normalize();
        Vector3 direction=forward*input.y+right*input.x;
        if(direction.sqrMagnitude>.01f)player.rotation=Quaternion.Slerp(player.rotation,Quaternion.LookRotation(direction),Time.deltaTime*14);
        if(motor.isGrounded && verticalSpeed<0)verticalSpeed=-2;
        verticalSpeed-=gravity*Time.deltaTime;
        CollisionFlags collisions=motor.Move((direction*(slowMode?baseMoveSpeed:moveSpeed)+Vector3.up*verticalSpeed)*Time.deltaTime);
        if((collisions & CollisionFlags.Above)!=0 && verticalSpeed>0)verticalSpeed=0;
        if(player.position.y< -12)Respawn();
        if(playerAnimator)
        {
            playerAnimator.SetFloat("Forward",input.magnitude);playerAnimator.SetFloat("MoveSpeed",motor.isGrounded?input.magnitude:1);playerAnimator.SetBool("OnGround",motor.isGrounded);
            playerAnimator.SetFloat("Jump",verticalSpeed);playerAnimator.SetFloat("JumpLeg",0);
            UpdateCarryAnimation();
        }
    }
    public void Jump()
    {
        if(paused || IsKnockedDown || training || EndCardVisible || (panel && panel.activeSelf))return;
        if(!motor || !motor.enabled || !motor.isGrounded || verticalSpeed>0)return;
        verticalSpeed=Mathf.Sqrt(2*Mathf.Max(.01f,gravity)*Mathf.Max(.1f,jumpHeight));
        if(playerAnimator){playerAnimator.SetBool("OnGround",false);playerAnimator.SetFloat("Jump",verticalSpeed);}
    }
    private void UpdateInteraction()
    {
        if(openingEgg){SetAction("",false);return;}
        nearby=null;float best=3.4f;
        foreach(Egg egg in eggs)if(!egg.banked && !egg.retrieved && egg!=carried && egg.transform)
        {float d=FlatDistance(player.position,egg.transform.position);if(d<best){best=d;nearby=egg;}}
        bool atHome=IsInsidePlayerCoop();
        if(IsCarrying && !carried.safe && player.position.z<=safePoint.position.z)GatherEgg();
        if(training)SetAction("STOP TRAINING",true);
        else if(pendingPet && atHome)SetAction("EQUIP PET",true);
        else if(IsCarrying && carried.safe && atHome && incubating==null)SetAction("INCUBATE",true);
        else if(IsCarrying)SetAction("BRING IT HOME",false);
        else if(incubating!=null && Time.time>=incubationEnd && atHome)SetAction("HATCH!",true);
        else if(trainingPoint && FlatDistance(player.position,trainingPoint.position)<3)SetAction("TRAIN",true);
        else if(nearby!=null && !pendingPet)SetAction("STEAL",true);
        else SetAction("",false);
        dropButton.gameObject.SetActive(IsCarrying && !carried.safe);
    }
    private void SetAction(string value,bool enabled)
    {
        actionTarget=value=="STEAL" && nearby!=null?nearby.transform:value=="HATCH!" && incubating!=null?incubating.transform:null;
        actionButton.gameObject.SetActive(originalActions.Count==0 && value!="");actionButton.interactable=enabled;actionText.text=value;
        foreach(var pair in originalActions)pair.Value.SetActive(pair.Key==value && enabled);
        if(!actionTarget)
        {
            if(originalFunctions)originalFunctions.anchoredPosition=Vector2.zero;
            if(actionButton)((RectTransform)actionButton.transform).anchoredPosition=Vector2.zero;
        }
    }
    public void Interact()
    {
        if(paused || IsKnockedDown || EndCardVisible || panel.activeSelf || openingEgg)return;
        if(training){training=false;return;}
        if(pendingPet && IsInsidePlayerCoop()){EquipPet();return;}
        if(IsCarrying && carried.safe && IsInsidePlayerCoop() && incubating==null){Incubate();return;}
        if(incubating!=null && Time.time>=incubationEnd && IsInsidePlayerCoop()){Hatch();return;}
        if(!IsCarrying && trainingPoint && FlatDistance(player.position,trainingPoint.position)<3){BeginTraining();return;}
        if(!IsCarrying && !pendingPet && nearby!=null && !nearby.banked && !nearby.retrieved)TakeEgg(nearby);
    }

    private void Incubate()
    {
        if(!IsInsidePlayerCoop())return;
        if(incubating!=null){Notify("Hatch the egg in your coop first!");return;}
        if(IsCarrying && carried.safe){incubating=carried;carried=null;}
        else return;

        UpdateCarryAnimation();

        PlaySound(deliverSound);
    }
    private bool IsInsidePlayerCoop()
    {
        if(!playerCoopArea || !player)return false;
        Vector3 local=playerCoopArea.transform.InverseTransformPoint(player.position)-playerCoopArea.center;
        Vector3 half=playerCoopArea.size*.5f;
        return Mathf.Abs(local.x)<half.x-.35f && Mathf.Abs(local.z)<half.z-.35f && Mathf.Abs(local.y)<=half.y+.2f;
    }
    private void PlaceEggOnCoopFloor(Egg egg)
    {
        Vector3 point=player.position+player.forward*1.2f;
        Vector3 local=playerCoopArea.transform.InverseTransformPoint(point);
        Vector3 center=playerCoopArea.center,half=playerCoopArea.size*.5f;
        local.x=Mathf.Clamp(local.x,center.x-half.x+1,center.x+half.x-1);
        local.z=Mathf.Clamp(local.z,center.z-half.z+1,center.z+half.z-1);
        local.y=center.y-half.y;
        point=playerCoopArea.transform.TransformPoint(local);
        RaycastHit hit;
        if(Physics.Raycast(point+Vector3.up*2,Vector3.down,out hit,4,~0,QueryTriggerInteraction.Ignore) && !hit.transform.IsChildOf(player))point.y=hit.point.y;
        egg.transform.rotation=egg.rotation;
        egg.transform.position=point;
        bool found=false;Bounds bounds=new Bounds();
        foreach(Renderer r in egg.transform.GetComponentsInChildren<Renderer>())
        {
            if(!r.enabled || r is ParticleSystemRenderer || r is LineRenderer)continue;
            if(!found){bounds=r.bounds;found=true;}else bounds.Encapsulate(r.bounds);
        }
        if(found)egg.transform.position+=Vector3.up*(point.y-bounds.min.y);
    }
    private void UpdateIncubation()
    {
        if(incubating==null){hatchText.text="";return;}
        if(openingEgg)
        {
            eggOpenClock+=Time.deltaTime;
            float progress=Mathf.Clamp01(eggOpenClock/Mathf.Max(.1f,eggOpenDuration));
            incubating.transform.rotation=eggOpenRotation*Quaternion.Euler(0,0,Mathf.Sin(progress*42)*Mathf.Lerp(4,22,progress));
            hatchText.text="";
            if(progress>=1)CompleteHatch();
            return;
        }
        float left=Mathf.Max(0,incubationEnd-Time.time);hatchText.text=left>0?Mathf.CeilToInt(left)+"s":"Open";
        incubating.transform.localRotation=Quaternion.Euler(0,Time.time*20,Mathf.Sin(Time.time*8)*3);
    }
    private void UpdateIncubationLabelPosition()
    {
        if(!hatchText)return;
        bool visible=incubating!=null && incubating.transform && !openingEgg && !EndCardVisible && !panel.activeSelf;
        if(!visible){hatchText.gameObject.SetActive(false);return;}
        Vector3 top=incubating.transform.position;
        foreach(Renderer renderer in incubating.transform.GetComponentsInChildren<Renderer>())
        {
            if(!renderer.enabled || renderer is ParticleSystemRenderer || renderer is LineRenderer)continue;
            top.y=Mathf.Max(top.y,renderer.bounds.max.y);
        }
        Vector3 screen=gameCamera.WorldToScreenPoint(top+Vector3.up*.35f);
        hatchText.gameObject.SetActive(screen.z>0);
        if(screen.z>0)hatchText.rectTransform.position=new Vector3(screen.x,screen.y,0);
    }
    private void Hatch()
    {
        if(openingEgg || incubating==null || Time.time<incubationEnd)return;
        openingEgg=true;eggOpenClock=0;eggOpenRotation=incubating.transform.rotation;
        hatchText.gameObject.SetActive(false);SetAction("",false);PlaySound(hatchSound);
    }
    private void CompleteHatch()
    {
        openingEgg=false;
        if(eggBreakEffect)
        {
            GameObject effect=Instantiate(eggBreakEffect,incubating.transform.position,Quaternion.identity);
            effect.transform.localScale=incubating.transform.localScale;
            effect.SetActive(true);
            foreach(ParticleSystem particles in effect.GetComponentsInChildren<ParticleSystem>(true))particles.Play();
            Destroy(effect,4);
        }
        Zone zone=incubating.zone;
        if(zone.petPrefabs.Length>0)
        {
            PlayableEggRules rule=Rule(incubating);
            int index=rule!=null?PlayableEggRules.Pick(rule.petWeights,rule.pets.Length):0;
            GameObject pet=Instantiate(rule!=null?rule.pets[index]:zone.petPrefabs[incubating.variant%zone.petPrefabs.Length]);DisablePhysics(pet);
            pet.transform.localScale*=incubating.size;
            pendingPetIncome=rule!=null?Mathf.Max(1,(int)(rule.petIncome[index]*Mathf.Pow(1.1f,incubating.size))):1;
            pendingPet=pet.transform;pendingPet.SetParent(carryPoint,false);pendingPet.localPosition=Vector3.zero;
        }
        Destroy(incubating.transform.gameObject);
        incubating.transform=null;incubating=null;
        Track("egg_hatched",Delivered);SetStep(TutorialStep.Equip);Notify("NEW PET! Place it in your coop to earn income.");
        UpdateCarryAnimation();
    }
    private void UpdateCarryAnimation()
    {
        if(!playerAnimator)return;
        int layer=playerAnimator.GetLayerIndex("Stealing");
        if(layer>=0)playerAnimator.SetLayerWeight(layer,IsCarrying && !carried.safe?1:0);
        int ownedLayer=playerAnimator.GetLayerIndex("OwnedCarry");
        if(ownedLayer>=0)playerAnimator.SetLayerWeight(ownedLayer,(IsCarrying && carried.safe) || pendingPet?1:0);
    }

        if(tutorialCompleted && pets.Count>=capturesForEndCard)ShowEndCard();
    }

        if(playerAnimator)playerAnimator.SetFloat("Forward",1);

        if(trainingClock>=.5f){trainingClock-=.5f;SpeedPower+=trainingPowerPerTick;moveSpeed=baseMoveSpeed+Mathf.Sqrt(SpeedPower)*speedPowerMultiplier;Notify("+"+trainingPowerPerTick+" SPEED POWER");
            if(Step==TutorialStep.Train && pets.Count>0){tutorialCompleted=true;inventoryBatSelected=true;SetStep(TutorialStep.Explore);Track("tutorial_completed",1);RefreshInventoryUI();Notify("TUTORIAL COMPLETE! BAT UNLOCKED.");}}
    }
    private void UpdateGuardians()
    {
        foreach(Zone zone in zones)
        {
            if(!zone.guardian)continue;
            if(hitZone==zone)continue;
            bool chasing=IsCarrying && !carried.safe && carried.zone==zone && Time.time>chaseDelay && Time.time>catchCooldown;
            Egg dropped=null;
            foreach(Egg egg in eggs)if(egg.zone==zone && egg.dropped && !egg.safe && !egg.retrieved){dropped=egg;break;}
            Vector3 destination=chasing?player.position:dropped!=null?dropped.transform.position:zone.guardianHome;destination.y=zone.guardian.position.y;
            float distance=FlatDistance(zone.guardian.position,destination);
            if(zone.warningMark)zone.warningMark.SetActive(chasing);
            if(zone.sleepingEffect)zone.sleepingEffect.SetActive(!chasing && dropped==null && distance<=.3f);
            if(distance>.3f)
            {
                zone.guardian.rotation=Quaternion.Slerp(zone.guardian.rotation,Quaternion.LookRotation(destination-zone.guardian.position),Time.deltaTime*6);
                float speed=chasing?(moveSpeed<zone.guardianSpeed || distance*distance>zone.attackDistance*zone.attackDistance*8?zone.guardianSpeed*2:moveSpeed):zone.guardianSpeed*4;
                zone.guardian.position=Vector3.MoveTowards(zone.guardian.position,destination,speed*Time.deltaTime);
            }
            else if(!chasing)zone.guardian.rotation=zone.guardianRotation;
            if(zone.animator)SetAnimatorMovement(zone.animator,distance>.3f?1:0);
            if(!chasing && dropped!=null && distance<1){dropped.retrieved=true;dropped.transform.gameObject.SetActive(false);}
            if(!chasing && dropped==null && distance<.4f)foreach(Egg egg in eggs)if(egg.zone==zone && egg.retrieved)RestoreEgg(egg);
            if(chasing && distance<zone.attackDistance && !IsKnockedDown)BeginKnockdown(zone);
        }
    }
    private void BeginKnockdown(Zone zone)
    {
        hitZone=zone;hitClock=0;hitApplied=false;training=false;joystick.ResetInput();
        hitDirection=player.position-zone.guardian.position;hitDirection.y=0;
        if(hitDirection.sqrMagnitude<.01f)hitDirection=zone.guardian.forward;
        hitDirection.Normalize();zone.guardian.rotation=Quaternion.LookRotation(hitDirection);attackerRotation=zone.guardian.rotation;
        if(playerAnimator){modelPosition=playerAnimator.transform.localPosition;modelRotation=playerAnimator.transform.localRotation;playerAnimator.SetFloat("Forward",0);}
        SetAction("",false);dropButton.gameObject.SetActive(false);
    }
    private void UpdateKnockdown()
    {
        hitClock+=Time.deltaTime;
        // The source guardian only has locomotion clips: use a short strike lean for the attack.
        float strike=Mathf.Sin(Mathf.Clamp01(hitClock/(hitWindup+.2f))*Mathf.PI)*24;
        hitZone.guardian.rotation=attackerRotation*Quaternion.Euler(strike,0,0);
        if(hitClock<hitWindup)return;
        if(!hitApplied)
        {
            hitApplied=true;DropEgg();PlaySound(caughtSound);Track("caught",1);
            if(ragdoll){motor.enabled=false;ragdoll.Begin(hitDirection*Mathf.Clamp(hitZone.guardianSpeed,1,30)*50);}
            if(playerAnimator)playerAnimator.enabled=false;
            Notify("OUCH! You dropped the egg.");verticalSpeed=3;
        }
        float elapsed=hitClock-hitWindup;
        if(ragdoll)
        {
            ragdoll.Follow();
            if(elapsed>=knockdownDuration)
            {
                ragdoll.Recover();motor.enabled=true;hitZone.guardian.rotation=attackerRotation;hitZone=null;catchCooldown=Time.time+2;verticalSpeed=0;
                SetStep(bag.Count>0?TutorialStep.Incubate:TutorialStep.FindEgg);
            }
            return;
        }
        float rise=Mathf.Clamp01((elapsed-knockdownDuration)/getUpDuration);
        float fall=Mathf.Clamp01(elapsed/.25f)*(1-rise);
        if(playerAnimator)
        {
            Vector3 localDirection=player.InverseTransformDirection(hitDirection);
            Vector3 axis=Vector3.Cross(Vector3.up,localDirection);
            playerAnimator.transform.localRotation=Quaternion.AngleAxis(85*fall,axis)*modelRotation;
            playerAnimator.transform.localPosition=modelPosition+Vector3.up*(.3f*fall);
        }
        if(motor.isGrounded && verticalSpeed<0)verticalSpeed=-2;
        verticalSpeed-=20*Time.deltaTime;
        motor.Move((hitDirection*(elapsed<.4f?5*(1-elapsed/.4f):0)+Vector3.up*verticalSpeed)*Time.deltaTime);
        if(rise>=1)
        {
            if(playerAnimator){playerAnimator.transform.localPosition=modelPosition;playerAnimator.transform.localRotation=modelRotation;playerAnimator.enabled=true;}
            hitZone.guardian.rotation=attackerRotation;hitZone=null;catchCooldown=Time.time+2;verticalSpeed=0;
            // Preserve player position: recovering from a hit never teleports to spawn.
            SetStep(bag.Count>0?TutorialStep.Incubate:TutorialStep.FindEgg);
        }
    }

    }

    private int RollEgg(Zone zone)
    {
        if(zone.eggRules==null || zone.eggRules.Length==0)return UnityEngine.Random.Range(0,zone.eggPrefabs.Length);
        float[] weights=new float[zone.eggRules.Length];for(int i=0;i<weights.Length;i++)weights[i]=zone.eggRules[i].spawnWeight;

    public static void DisablePhysics(GameObject obj)
    {
        foreach(Collider c in obj.GetComponentsInChildren<Collider>(true))c.enabled=false;
        foreach(Rigidbody body in obj.GetComponentsInChildren<Rigidbody>(true)){body.isKinematic=true;body.useGravity=false;}
        foreach(Transform t in obj.GetComponentsInChildren<Transform>(true))if(t.name.ToLowerInvariant().Contains("highlight"))t.gameObject.SetActive(false);
        foreach(TMPro.TMP_Text label in obj.GetComponentsInChildren<TMPro.TMP_Text>(true))label.gameObject.SetActive(false);
    }
    public static void SetVisualSize(GameObject obj,float height)
    {
        bool found=false;Bounds bounds=new Bounds();
        foreach(Renderer r in obj.GetComponentsInChildren<Renderer>())if(!(r is ParticleSystemRenderer) && !(r is LineRenderer))
        {if(!found){bounds=r.bounds;found=true;}else bounds.Encapsulate(r.bounds);}
        if(found && bounds.size.y>.01f)obj.transform.localScale*=height/bounds.size.y;
    }
    private static void SetAnimatorMovement(Animator a,float value)
    {
        foreach(AnimatorControllerParameter p in a.parameters)
        {
            if(p.type==AnimatorControllerParameterType.Float && (p.name=="Forward" || p.name=="Speed" || p.name=="MoveSpeed"))a.SetFloat(p.name,value);
            if(p.type==AnimatorControllerParameterType.Bool && (p.name=="IsMoving" || p.name=="Run"))a.SetBool(p.name,value>0);
        }
    }
