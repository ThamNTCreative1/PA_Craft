    private string UIAsset(string name){return "Assets/MiniGames/StealAnEgg/Sprites/UIs/"+name;}
    private Button AuthorInventorySlot(Transform parent,string name,Vector2 anchor,Vector2 position,float size)
    {
        Button slot=ReferenceIcon(parent,name,UIAsset("SAE_Inventory_Slot.png"),anchor,size,"");
        ((RectTransform)slot.transform).anchoredPosition=position;
        Image icon=ReferenceImage(slot.transform,"Icon",UIAsset("sae_egg (2).png"));
        Place(icon.rectTransform,new Vector2(.06f,.06f),new Vector2(.94f,.94f));icon.gameObject.SetActive(false);
        Label("Label",slot.transform,"",14,new Vector2(.02f,.02f),new Vector2(.98f,.23f));

        return slot;
    }
    private void ReferencePlace(RectTransform rt,Vector2 anchor,Vector2 position,Vector2 size)
    {rt.anchorMin=rt.anchorMax=anchor;rt.pivot=new Vector2(.5f,.5f);rt.anchoredPosition=position;rt.sizeDelta=size;rt.localScale=Vector3.one;}
    private Image ReferenceImage(Transform parent,string name,string path)
    {RectTransform rt=Rect(name,parent,Vector2.zero,Vector2.one);Image img=rt.gameObject.AddComponent<Image>();img.sprite=AssetDatabase.LoadAssetAtPath<Sprite>(path);img.preserveAspect=true;img.raycastTarget=false;return img;}
    private Button ReferenceIcon(Transform parent,string name,string path,Vector2 anchor,float size,string caption)
    {
        Image img=ReferenceImage(parent,name,path);ReferencePlace(img.rectTransform,anchor,Vector2.zero,new Vector2(size,size));
        Button b=img.gameObject.AddComponent<Button>();b.targetGraphic=img;img.raycastTarget=true;
        if(caption!="")Label("Label",img.transform,caption,26,new Vector2(-.45f,-.25f),new Vector2(1.45f,.15f));
        return b;
    }
    private void ReferenceCounter(Transform template,Transform parent,string name,Vector2 anchor,Sprite sprite,Color color)
