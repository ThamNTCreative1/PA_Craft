    private void ApplySourceArt()
    {
        if(!sourceHud)return;
        GameObject art=Instantiate(sourceHud,safeUI,false);art.name="Original game HUD";art.SetActive(true);
        Place((RectTransform)art.transform,Vector2.zero,Vector2.one);
        Transform[] all=art.GetComponentsInChildren<Transform>(true);
        foreach(Image img in art.GetComponentsInChildren<Image>(true))img.raycastTarget=false;
        foreach(Transform t in all)
        {
            if(t.name=="EquipPetPopup" || t.name=="IncubatingEggPopup" || t.name=="UIBoost" || t.name=="TimeOfDay Frame" || t.name=="ZoneEnterHighlight" || t.name=="Right" || t.name=="Button_Attack" || t.name=="Button_Setting")t.gameObject.SetActive(false);
            if(t.name=="[Container] FuntionButtons")originalFunctions=(RectTransform)t;
            if(t.name=="MoneyFrame")originalMoney=BindCounter(t,moneyIcon);
            if(t.name=="SpeedPowerFrame")originalSpeed=BindCounter(t,speedIcon);
            if(t.parent && t.parent.name=="Left")
            {
                // Keep the original hierarchy and placement; attach standalone callbacks.
                if(t.name=="Button_Index")BindButton(t,()=>OpenPanel("PETS"));
                else if(t.name=="Button_IAPShop")BindButton(t,()=>OpenPanel("SHOP"));
                else if(t.name=="Button_ShopTrail")BindButton(t,UpgradeSpeed);
                else if(t.name=="[Toggle] Speed Mode")BindButton(t,()=>{slowMode=!slowMode;Notify(slowMode?"SLOW MODE":"FAST MODE");});
            }
        }
        foreach(Transform t in all)
        {
            if(t.parent==originalFunctions)
            {
                string key=t.name=="[Button] Steal"?"STEAL":t.name=="[Button] Incubate"?"INCUBATE":t.name=="[Button] Hatch"?"HATCH!":t.name=="[Button] Equip Monster"?"EQUIP PET":t.name=="[Button] Use Treadmil"?"TRAIN":"";
                t.gameObject.SetActive(false);
                if(key!=""){originalActions[key]=t.gameObject;BindButton(t,Interact);}
                if(t.name=="[Button] DropStealingEgg")
                {
                    dropButton.gameObject.SetActive(false);dropButton=BindButton(t,DropEgg);
                }
            }
        }
        foreach(string name in new[]{"Coins","Progress","Zones","Pets","Speed","Shop"}){Transform old=safeUI.Find(name);if(old)old.gameObject.SetActive(false);}
        // The tutorial and joystick remain above the original HUD; end card is ad-specific.
        safeUI.Find("Tutorial").gameObject.SetActive(false);
        safeUI.Find("Controls").gameObject.SetActive(false);
        safeUI.Find("Joystick").SetAsLastSibling();
        panel.transform.SetAsLastSibling();endCard.transform.SetAsLastSibling();
    }
    private Button BindButton(Transform target,UnityEngine.Events.UnityAction action)
    {
        Button b=target.GetComponent<Button>();if(!b)b=target.gameObject.AddComponent<Button>();
        Graphic graphic=target.GetComponent<Graphic>();if(graphic){graphic.raycastTarget=true;b.targetGraphic=graphic;}
        b.onClick=new Button.ButtonClickedEvent();b.onClick.AddListener(action);return b;
    }
    private TMPro.TMP_Text BindCounter(Transform target,Sprite icon)
    {
        foreach(Image img in target.GetComponentsInChildren<Image>(true))if(img.name=="Image"){img.sprite=icon;img.enabled=icon!=null;}
        return target.GetComponentInChildren<TMPro.TMP_Text>(true);
    }
    private void CopyControlArt(Transform[] all,string sourceName,string targetName,UnityEngine.Events.UnityAction action)
    {
        foreach(Transform t in all)if(t.name==sourceName)
        {
            RectTransform target=(RectTransform)safeUI.Find(targetName);
            t.SetParent(safeUI,false);Place((RectTransform)t,target.anchorMin,target.anchorMax);t.gameObject.SetActive(true);
            Button b=t.GetComponent<Button>();if(!b)b=t.gameObject.AddComponent<Button>();b.onClick=new Button.ButtonClickedEvent();b.onClick.AddListener(action);
            if(targetName=="Speed")foreach(TMPro.TMP_Text label in t.GetComponentsInChildren<TMPro.TMP_Text>(true))label.text="Speed +";
            target.gameObject.SetActive(false);return;
        }
    }
